CREATE TABLE PROJ.DEFENSOR
( ID_DEFENSOR INT AUTO_INCREMENT PRIMARY KEY COMMENT 'Identificador único do Defensor',
  ID_PERMISSIONARIO INT NOT NULL COMMENT 'Chave estrangeira com identificador do Permissionário',
  NOME_DEFENSOR VARCHAR(150) NOT NULL COMMENT 'Nome do Defensor',
  CPF_DEFENSOR VARCHAR(14) COMMENT 'CPF do Defensor',
  RG_DEFENSOR VARCHAR(12) COMMENT 'RG do Defensor',
  ORGAO_EMISSOR VARCHAR(10) COMMENT 'Órgão que emitiu o RG/Identidade do Defensor',
  SEXO VARCHAR(1) NOT NULL COMMENT 'Sexo do Defensor',
  ESTADO_CIVIL VARCHAR(2) COMMENT 'Estado Civil do Defensor',
  DATA_NASCIMENTO DATE NOT NULL COMMENT 'Data de Nascimento do Defensor',
  FILIACAO_MAE VARCHAR(100) COMMENT 'Filiação mãe do Defensor',
  FILIACAO_PAI VARCHAR(100) COMMENT 'Filiação pai do Defensor',
  UF_DEFENSOR VARCHAR(2) NOT NULL COMMENT 'UF do Defensor',
  CIDADE_DEFENSOR VARCHAR(150) NOT NULL COMMENT 'Cidade do Defensor',
  BAIRRO_DEFENSOR VARCHAR(150) NOT NULL COMMENT 'Bairro do Defensor',
  CEP VARCHAR(10) NOT NULL COMMENT 'CEP do Defensor',
  ENDERECO_DEFENSOR VARCHAR(150) NOT NULL COMMENT 'Endereço do Defensor',
  COMPROVANTE_RESIDENCIA MEDIUMBLOB NOT NULL COMMENT 'Anexo do Comprovante de Residência do Defensor',
  CELULAR_DEFENSOR VARCHAR(12) COMMENT 'Nº do celular/whatsapp do Defensor',
  EMAIL_DEFENSOR VARCHAR(50) COMMENT 'E-mail do Defensor',
  CNH_DEFENSOR VARCHAR(15) NOT NULL COMMENT 'CNH do Defensor',
  CATEGORIA_CNH_DEFENSOR VARCHAR(2) COMMENT 'Categoria da CNH do Defensor',
  DATA_VALIDADE_CNH DATE COMMENT 'Data de Validade da CNH do Defensor',
  NUMERO_QUITACAO_MILITAR VARCHAR(15) COMMENT 'Nº da quitação militar do Defensor',
  NUMERO_QUITACAO_ELEITORAL VARCHAR(15) COMMENT 'Nº da quitação eleitoral do Defensor',
  NUMERO_INSCRICAO_INSS VARCHAR(15) COMMENT 'N] da inscrição do INSS do Defensor',
  NUMERO_CERTIFICADO_CONDUTOR VARCHAR(15) COMMENT 'Número do certificado de condutor',
  DATA_VALIDADE_CER_CON DATE COMMENT 'Data de Validade do certificado de condutor',
  ANEXO_RG MEDIUMBLOB NOT NULL COMMENT 'Anexo do RG do Defensor',
  ANEXO_CPF MEDIUMBLOB NOT NULL COMMENT 'Anexo do CPF do Defensor',
  ANEXO_CNH MEDIUMBLOB NOT NULL COMMENT 'Anexo da CNH do Defensor',
  CERTIFICADO_CONDUTOR MEDIUMBLOB NOT NULL COMMENT 'Anexo do Certificado de Condutor do Defensor',
  CERTIDAO_NEGATIVA_CRIMINAL MEDIUMBLOB NOT NULL COMMENT 'Anexo da Certidão Negativa Criminal do Defensor',
  CERTIDAO_NEGATIVA_MUNICIPAL MEDIUMBLOB NOT NULL COMMENT 'Anexo da Certidão Negativa Municipal  do Defensor',
  CERTIFICADO_PROPRIEDADE MEDIUMBLOB NOT NULL COMMENT 'Anexo do Certificado de Propriedade do Defensor',
  APOLICE_SEGURO_VIDA MEDIUMBLOB NOT NULL COMMENT 'Anexo da Apólice de Seguro de Vida do Defensor',
  APOLICE_SEGURO_MOTOCICLETA MEDIUMBLOB NOT NULL COMMENT 'Anexo da Apólice de Seguro da Motocicleta do Defensor',
  FOTO MEDIUMBLOB NOT NULL COMMENT 'Anexo da Foto do Defensor',
  DATA_CRIACAO DATE COMMENT 'Data de criação do Defensor',
  STATUS VARCHAR(10) COMMENT 'Indica o Status do Defensor: 1-Ativo; 2-Inativo, 3-Suspenso; 4-Cassado',
  APLICATIVO_ALTERNATIVO VARCHAR(2) COMMENT 'Indica se o Defensor é de Aplicativo Alternativo',
  OBSERVACAO VARCHAR(200) COMMENT 'Observação com dados gerais do Defensor'
);
