CREATE TABLE PROJ.DEFENSOR
( ID_DEFENSOR NUMBER DEFAULT PROJ.SEQ_DEFENSOR.NEXTVAL,
  NUMERO_PERMISSAO VARCHAR2(50) NOT NULL,
  NOME_DEFENSOR VARCHAR2(150) NOT NULL,
  CPF_DEFENSOR VARCHAR2(11),
  CNPJ_EMPRESA VARCHAR2(14),
  RG_DEFENSOR VARCHAR2(10),
  ORGAO_EMISSOR VARCHAR2(10),
  NATUREZA_PESSOA VARCHAR2(1) NOT NULL,
  UF_DEFENSOR VARCHAR2(2) NOT NULL,
  CIDADE_DEFENSOR VARCHAR2(150) NOT NULL,
  BAIRRO_DEFENSOR VARCHAR2(150) NOT NULL,
  ENDERECO_DEFENSOR VARCHAR2(150) NOT NULL,
  CELULAR_DEFENSOR VARCHAR2(12),
  CNH_DEFENSOR VARCHAR2(15) NOT NULL,
  CATEGORIA_CNH_DEFENSOR VARCHAR2(2),
  NUMERO_QUITACAO_MILITAR VARCHAR2(15),
  NUMERO_QUITACAO_ELEITORAL VARCHAR2(15),
  NUMERO_INSCRICAO_INSS VARCHAR2(15),
  NUMERO_CERTIFICADO_CONDUTOR VARCHAR2(15),
  CERTIFICADO_CONDUTOR BLOB NOT NULL,
  CERTIDAO_NEGATIVA_CRIMINAL BLOB NOT NULL,
  CERTIDAO_NEGATIVA_MUNICIPAL BLOB NOT NULL,
  FOTO BLOB NOT NULL,
  DATA_CRIACAO DATE,
  STATUS VARCHAR2(10),
  CONSTRAINT DEFENSOR_PK PRIMARY KEY (ID_DEFENSOR)
);
-- Add comments to the columns 
COMMENT ON COLUMN PROJ.DEFENSOR.ID_DEFENSOR IS 'Identificador único do Defensor';
COMMENT ON COLUMN PROJ.DEFENSOR.NUMERO_PERMISSAO  IS 'Número da Permissão';
COMMENT ON COLUMN PROJ.DEFENSOR.NOME_DEFENSOR  IS 'Nome do Defensor';
COMMENT ON COLUMN PROJ.DEFENSOR.CPF_DEFENSOR  IS 'CPF do Defensor';
COMMENT ON COLUMN PROJ.DEFENSOR.CNPJ_EMPRESA  IS 'CNPJ da Empresa do Defensor';
COMMENT ON COLUMN PROJ.DEFENSOR.RG_DEFENSOR  IS 'RG do Defensor';
COMMENT ON COLUMN PROJ.DEFENSOR.ORGAO_EMISSOR  IS 'Órgão que emitiu o RG/Identidade do Defensor';
COMMENT ON COLUMN PROJ.DEFENSOR.NATUREZA_PESSOA  IS 'Natureza da Pessoa';
COMMENT ON COLUMN PROJ.DEFENSOR.UF_DEFENSOR  IS 'UF do Defensor';
COMMENT ON COLUMN PROJ.DEFENSOR.BAIRRO_DEFENSOR IS 'Bairro do Defensor';
COMMENT ON COLUMN PROJ.DEFENSOR.ENDERECO_DEFENSOR  IS 'Endereço do Defensor';
COMMENT ON COLUMN PROJ.DEFENSOR.CELULAR_DEFENSOR  IS 'Celular do Defensor';
COMMENT ON COLUMN PROJ.DEFENSOR.CNH_DEFENSOR  IS 'Habilitação do Defensor';
COMMENT ON COLUMN PROJ.DEFENSOR.CATEGORIA_CNH_DEFENSOR  IS 'Categoria da CNH do Defensor';
COMMENT ON COLUMN PROJ.DEFENSOR.NUMERO_QUITACAO_MILITAR  IS 'Número da quitação militar';
COMMENT ON COLUMN PROJ.DEFENSOR.NUMERO_QUITACAO_ELEITORAL  IS 'Número da quitação eleitoral';
COMMENT ON COLUMN PROJ.DEFENSOR.NUMERO_INSCRICAO_INSS  IS 'Número da inscrição no INSS';
COMMENT ON COLUMN PROJ.DEFENSOR.NUMERO_CERTIFICADO_CONDUTOR  IS 'Número do certificado de condutor';
COMMENT ON COLUMN PROJ.DEFENSOR.CERTIDAO_NEGATIVA_CRIMINAL  IS 'Certidão Negativa Criminal do Defensor';
COMMENT ON COLUMN PROJ.DEFENSOR.CERTIDAO_NEGATIVA_MUNICIPAL  IS 'Certidão Negativa Municipal do Defensor';
COMMENT ON COLUMN PROJ.DEFENSOR.FOTO  IS 'Foto 3x4 do Defensor';
COMMENT ON COLUMN PROJ.DEFENSOR.DATA_CRIACAO  IS 'Data de criação do Defensor';
