CREATE TABLE PROJ.VEICULO
( ID_VEICULO NUMBER DEFAULT PROJ.SEQ_VEICULO.NEXTVAL,
  ID_PERMISSIONARIO NUMBER NOT NULL,	
  ID_PONTOS_TAXI NUMBER NOT NULL,
  NUMERO_PERMISSAO VARCHAR2(20) NOT NULL,  
  PLACA VARCHAR2(7) NOT NULL,
  RENAVAM VARCHAR2(11) NOT NULL,
  CHASSI VARCHAR2(17) NOT NULL,
  ANO_FABRICACAO VARCHAR2(4) NOT NULL,
  MARCA VARCHAR2(50) NOT NULL,
  MODELO VARCHAR2(50) NOT NULL,
  ANO_MODELO VARCHAR2(4) NOT NULL,
  COR VARCHAR2(2) NOT NULL,
  COMBUSTIVEL VARCHAR2(2) NOT NULL,
  CRLV BLOB NOT NULL,
  NUMERO_TAXIMETRO VARCHAR2(20) NOT NULL,
  ANO_RENOVACAO VARCHAR2(4) NOT NULL,
  DATA_VISTORIA DATE,
  DATA_RETORNO DATE,
  COMPROVANTE_VISTORIA BLOB NOT NULL,
  SITUACAO_VEICULO VARCHAR2(50),
  NUMERO_CRLV VARCHAR2(20),
  ANO_CRLV VARCHAR2(4),
  CERTIFICADO_AFERICAO VARCHAR2(15),
  TIPO_VEICULO VARCHAR2(2) NOT NULL,
  OBSERVACAO VARCHAR2(200),
  DATA_CRIACAO DATE,
  STATUS VARCHAR2(10),
  CONSTRAINT VEICULO_PK PRIMARY KEY (ID_VEICULO)
);
-- Add comments to the columns 
COMMENT ON COLUMN PROJ.VEICULO.ID_VEICULO IS 'Identificador único do veículo';
COMMENT ON COLUMN PROJ.VEICULO.ID_PERMISSIONARIO IS 'Identificador chave estrangeira do permissionário do veículo';
COMMENT ON COLUMN PROJ.VEICULO.ID_PONTOS_TAXI IS 'Identificador chave estrangeira do ponto de táxi do veículo';
COMMENT ON COLUMN PROJ.VEICULO.NUMERO_PERMISSAO  IS 'Número da Permissão do veículo';
COMMENT ON COLUMN PROJ.VEICULO.PLACA  IS 'Placa do veículo';
COMMENT ON COLUMN PROJ.VEICULO.RENAVAM  IS 'Renavam do veículo';
COMMENT ON COLUMN PROJ.VEICULO.CHASSI  IS 'Chassi do veículo';
COMMENT ON COLUMN PROJ.VEICULO.ANO_FABRICACAO  IS 'Ano Fabricação do veículo';
COMMENT ON COLUMN PROJ.VEICULO.MARCA  IS 'Marca do veículo';
COMMENT ON COLUMN PROJ.VEICULO.MODELO  IS 'Modelo do veículo';
COMMENT ON COLUMN PROJ.VEICULO.ANO_MODELO  IS 'Modelo do veículo';
COMMENT ON COLUMN PROJ.VEICULO.COR  IS 'Cor do veículo';
COMMENT ON COLUMN PROJ.VEICULO.COMBUSTIVEL  IS 'Combustível do veículo';
COMMENT ON COLUMN PROJ.VEICULO.CRLV  IS 'Documento Certificado de Registro e Licenciamento de Veículo';
COMMENT ON COLUMN PROJ.VEICULO.NUMERO_TAXIMETRO  IS 'Número do Taxímetro do veículo';
COMMENT ON COLUMN PROJ.VEICULO.ANO_RENOVACAO  IS 'Ano de Renovação da Permissão';
COMMENT ON COLUMN PROJ.VEICULO.DATA_VISTORIA  IS 'Data de vistoria do veículo';
COMMENT ON COLUMN PROJ.VEICULO.DATA_RETORNO  IS 'Data de retorno para a vistoria';
COMMENT ON COLUMN PROJ.VEICULO.COMPROVANTE_VISTORIA  IS 'Documento de comprovação da vistoria';
COMMENT ON COLUMN PROJ.VEICULO.SITUACAO_VEICULO  IS 'Situação do veículo';
COMMENT ON COLUMN PROJ.VEICULO.NUMERO_CRLV  IS 'Número do Documento CRLV';
COMMENT ON COLUMN PROJ.VEICULO.ANO_CRLV  IS 'Ano de Emissão do CRLV';
COMMENT ON COLUMN PROJ.VEICULO.CERTIFICADO_AFERICAO  IS 'Certificado de Aferição do Veículo';
COMMENT ON COLUMN PROJ.VEICULO.TIPO_VEICULO  IS 'Indica o tipo do Veículo';
COMMENT ON COLUMN PROJ.VEICULO.OBSERVACAO  IS 'Observações gerais sobre o veículo';
COMMENT ON COLUMN PROJ.VEICULO.DATA_CRIACAO  IS 'Data de criação';
