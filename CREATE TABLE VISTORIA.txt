CREATE TABLE PROJ.VISTORIA
( ID_VISTORIA NUMBER DEFAULT PROJ.SEQ_VISTORIA.NEXTVAL,
  ID_VEICULO NUMBER NOT NULL, 
  CHASSIFUNILARIAPINTURA VARCHAR2(1),
  INSTALACAOELETRICA VARCHAR2(1),
  FAROLATLABAIXA VARCHAR2(1),
  BUZINA VARCHAR2(1),
  LANTERNATRASEIRA VARCHAR2(1),
  FREIODIANTEIRO VARCHAR2(1),
  LUZPLACA VARCHAR2(1),
  FREIOTRASEIRO VARCHAR2(1),
  LUZESDIRECAO VARCHAR2(1),
  PNEUSDESGATECALIBRAGEM VARCHAR2(1),
  LUZFREIO VARCHAR2(1),
  CORRENTECORREIA VARCHAR2(1),
  PLACASDIANTEIRATRASEIRA VARCHAR2(1),
  VAZAMENTOOLEOCOMBUSTIVEL VARCHAR2(1),
  LIMPEZAGERALINTERNA VARCHAR2(1),
  ESCAPAMENTO VARCHAR2(1),
  ASSENTOFIXACAO VARCHAR2(1),
  EQUIPAMENTOSOBRIGATORIOS VARCHAR2(1),
  ESPELHOSRETROVISORES VARCHAR2(1),
  SELOSVISTORIA VARCHAR2(1),
  GUIDAOMANOPLAS VARCHAR2(1),
  OUTROS VARCHAR2(1),  
  DATA_VISTORIA DATE NOT NULL,
  DATA_RETORNO DATE,
  STATUS_VISTORIA VARCHAR2(2) NOT NULL,
  RESSALVAS VARCHAR2(100),
  COMPROVANTE_VISTORIA BLOB NOT NULL,
  OBSERVACAO VARCHAR2(200),
  DATA_CRIACAO DATE NOT NULL,
  STATUS VARCHAR2(10) NOT NULL,
  CONSTRAINT VISTORIA_PK PRIMARY KEY (ID_VISTORIA)
);
-- Add comments to the columns 
COMMENT ON COLUMN PROJ.VISTORIA.ID_VISTORIA IS 'Identificador único da Vistoria do Veículo';
COMMENT ON COLUMN PROJ.VISTORIA.ID_VEICULO IS 'Identificador chave estrangeira do Veículo';
COMMENT ON COLUMN PROJ.VISTORIA.DATA_CRIACAO IS 'Data de Criaçã da Vistoria do Veículo';
