CREATE TABLE PROJ.PONTOS_TAXI
( ID_PONTOS_TAXI NUMBER DEFAULT PROJ.SEQ_PONTOS_TAXI.NEXTVAL,
  NUMERO_PONTO VARCHAR2(10) NOT NULL,
  DESCRICAO_PONTO VARCHAR2(150) NOT NULL,
  FATOR_ROTATIVIDADE VARCHAR2(10) NULL,
  REFERENCIA_PONTO VARCHAR2(100) NULL,
  NUMERO_VAGAS VARCHAR2(2) NOT NULL,
  MODALIDADE VARCHAR2(2) NOT NULL,
  DATA_CRIACAO DATE,
  STATUS VARCHAR2(10),
  CONSTRAINT PONTOS_TAXI_PK PRIMARY KEY (ID_PONTOS_TAXI)
);
-- Add comments to the columns 
COMMENT ON COLUMN PROJ.PONTOS_TAXI.ID_PONTOS_TAXI IS 'Identificador único do Ponto de Táxi';
COMMENT ON COLUMN PROJ.PONTOS_TAXI.NUMERO_PONTO  IS 'Número do Ponto de Táxi';
COMMENT ON COLUMN PROJ.PONTOS_TAXI.DESCRICAO_PONTO  IS 'Descrição do Ponto de Táxi';
COMMENT ON COLUMN PROJ.PONTOS_TAXI.FATOR_ROTATIVIDADE  IS 'Fator de Rotatividade do Ponto de Táxi';
COMMENT ON COLUMN PROJ.PONTOS_TAXI.REFERENCIA_PONTO  IS 'Ponto de Referência';
COMMENT ON COLUMN PROJ.PONTOS_TAXI.NUMERO_VAGAS  IS 'Número de Vagas do Ponto de Táxi';
COMMENT ON COLUMN PROJ.PONTOS_TAXI.MODALIDADE  IS 'Modalidade do Ponto de Táxi';
COMMENT ON COLUMN PROJ.PONTOS_TAXI.DATA_CRIACAO  IS 'Data de criação do Ponto de Táxi';
