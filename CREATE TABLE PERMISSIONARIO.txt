CREATE TABLE PROJ.PERMISSIONARIO
( ID_PERMISSIONARIO INT AUTO_INCREMENT PRIMARY KEY COMMENT 'Identificador único do permissionário',
  NUMERO_PERMISSAO VARCHAR(50) COMMENT 'Número da Permissão',
  NOME_PERMISSIONARIO VARCHAR(150) NOT NULL COMMENT 'Nome do Permissionário',
  CPF_PERMISSIONARIO VARCHAR(14) COMMENT 'CPF do Permissionário',
  RG_PERMISSIONARIO VARCHAR(12) COMMENT 'RG do Permissionário',
  ORGAO_EMISSOR VARCHAR(10) COMMENT 'Órgão que emitiu o RG/Identidade do Permissionário',
  SEXO VARCHAR(1) NOT NULL COMMENT 'Sexo do Permissionário',
  ESTADO_CIVIL VARCHAR(2) COMMENT 'Estado Civil do Permissionário',
  DATA_NASCIMENTO DATE NOT NULL COMMENT 'Data de Nascimento do Permissionário',
  FILIACAO_MAE VARCHAR(100) COMMENT 'Filiação mãe do Permissionário',
  FILIACAO_PAI VARCHAR(100) COMMENT 'Filiação pai do Permissionário',
  UF_PERMISSIONARIO VARCHAR(2) NOT NULL COMMENT 'UF do Permissionário',
  CIDADE_PERMISSIONARIO VARCHAR(150) NOT NULL COMMENT 'Cidade do Permissionário',
  BAIRRO_PERMISSIONARIO VARCHAR(150) NOT NULL COMMENT 'Bairro do Permissionário',
  CEP VARCHAR(10) NOT NULL COMMENT 'CEP do Permissionário',
  ENDERECO_PERMISSIONARIO VARCHAR(150) NOT NULL COMMENT 'Endereço do Permissionário',
  COMPROVANTE_RESIDENCIA MEDIUMBLOB NOT NULL COMMENT 'Anexo do Comprovante de Residência do Permissionário',
  CELULAR_PERMISSIONARIO VARCHAR(12) COMMENT 'Nº do celular/whatsapp do Permissionário',
  EMAIL_PERMISSIONARIO VARCHAR(50) COMMENT 'E-mail do Permissionário',
  CNH_PERMISSIONARIO VARCHAR(15) NOT NULL COMMENT 'CNH do Permissionário',
  CATEGORIA_CNH_PERMISSIONARIO VARCHAR(2) COMMENT 'Categoria da CNH do Permissionário',
  DATA_VALIDADE_CNH DATE COMMENT 'Data de Validade da CNH do Permissionário',
  NUMERO_QUITACAO_MILITAR VARCHAR(15) COMMENT 'Nº da quitação militar do Permissionário',
  NUMERO_QUITACAO_ELEITORAL VARCHAR(15) COMMENT 'Nº da quitação eleitoral do Permissionário',
  NUMERO_INSCRICAO_INSS VARCHAR(15) COMMENT 'N] da inscrição do INSS do Permissionário',
  NUMERO_CERTIFICADO_CONDUTOR VARCHAR(15) COMMENT 'Número do certificado de condutor',
  DATA_VALIDADE_CER_CON DATE COMMENT 'Data de Validade do certificado de condutor',
  ANEXO_RG MEDIUMBLOB NOT NULL COMMENT 'Anexo do RG do Permissionário',
  ANEXO_CPF MEDIUMBLOB NOT NULL COMMENT 'Anexo do CPF do Permissionário',
  ANEXO_CNH MEDIUMBLOB NOT NULL COMMENT 'Anexo da CNH do Permissionário',
  CERTIFICADO_CONDUTOR MEDIUMBLOB NOT NULL COMMENT 'Anexo do Certificado de Condutor do Permissionário',
  CERTIDAO_NEGATIVA_CRIMINAL MEDIUMBLOB NOT NULL COMMENT 'Anexo da Certidão Negativa Criminal do Permissionário',
  CERTIDAO_NEGATIVA_MUNICIPAL MEDIUMBLOB NOT NULL COMMENT 'Anexo da Certidão Negativa Municipal  do Permissionário',
  CERTIFICADO_PROPRIEDADE MEDIUMBLOB NOT NULL COMMENT 'Anexo do Certificado de Propriedade do Permissionário',
  APOLICE_SEGURO_VIDA MEDIUMBLOB NOT NULL COMMENT 'Anexo da Apólice de Seguro de Vida do Permissionário',
  APOLICE_SEGURO_MOTOCICLETA MEDIUMBLOB NOT NULL COMMENT 'Anexo da Apólice de Seguro da Motocicleta do Permissionário',
  FOTO MEDIUMBLOB NOT NULL COMMENT 'Anexo da Foto do Permissionário',
  DATA_CRIACAO DATE COMMENT 'Data de criação do Permissionário',
  STATUS VARCHAR(10) COMMENT 'Indica o Status do Permissionário: 1-Ativo; 2-Inativo, 3-Suspenso; 4-Cassado',
  APLICATIVO_ALTERNATIVO VARCHAR(2) COMMENT 'Indica se o Permissionário é de Aplicativo Alternativo',
  OBSERVACAO VARCHAR(200) COMMENT 'Observação com dados gerais do Permissionário'
);
