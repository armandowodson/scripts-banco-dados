CREATE TABLE PROJ.PERMISSIONARIO
( ID_PERMISSIONARIO NUMBER DEFAULT PROJ.SEQ_PERMISSIONARIO.NEXTVAL,
  NUMERO_PERMISSAO VARCHAR2(50) NOT NULL,
  NOME_PERMISSIONARIO VARCHAR2(150) NOT NULL,
  CPF_PERMISSIONARIO VARCHAR2(11),
  RG_PERMISSIONARIO VARCHAR2(10),
  ORGAO_EMISSOR VARCHAR2(10),
  SEXO VARCHAR2(1) NOT NULL,
  ESTADO_CIVIL VARCHAR2(2),
  DATA_NASCIMENTO DATE NOT NULL,
  UF_PERMISSIONARIO VARCHAR2(2) NOT NULL,
  CIDADE_PERMISSIONARIO VARCHAR2(150) NOT NULL,
  BAIRRO_PERMISSIONARIO VARCHAR2(150) NOT NULL,
  ENDERECO_PERMISSIONARIO VARCHAR2(150) NOT NULL,
  CELULAR_PERMISSIONARIO VARCHAR2(12),
  EMAIL_PERMISSIONARIO VARCHAR2(50),
  CNH_PERMISSIONARIO VARCHAR2(15) NOT NULL,
  CATEGORIA_CNH_PERMISSIONARIO VARCHAR2(2),
  NUMERO_QUITACAO_MILITAR VARCHAR2(15),
  NUMERO_QUITACAO_ELEITORAL VARCHAR2(15),
  NUMERO_INSCRICAO_INSS VARCHAR2(15),
  NUMERO_CERTIFICADO_CONDUTOR VARCHAR2(15),
  CERTIFICADO_CONDUTOR BLOB NOT NULL,
  CERTIDAO_NEGATIVA_CRIMINAL BLOB NOT NULL,
  CERTIDAO_NEGATIVA_MUNICIPAL BLOB NOT NULL,
  FOTO BLOB NOT NULL,
  DATA_CRIACAO DATE,
  STATUS VARCHAR2(10),
  APLICATIVO_ALTERNATIVO VARCHAR2(2),
  OBSERVACAO VARCHAR2(200),
  CONSTRAINT PERMISSIONARIO_PK PRIMARY KEY (ID_PERMISSIONARIO)
);
-- Add comments to the columns 
COMMENT ON COLUMN PROJ.PERMISSIONARIO.ID_PERMISSIONARIO IS 'Identificador único do Permissionário';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.NUMERO_PERMISSAO  IS 'Número da Permissão';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.NOME_PERMISSIONARIO  IS 'Nome do Permissionário';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.CPF_PERMISSIONARIO  IS 'CPF do Permissionário';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.RG_PERMISSIONARIO  IS 'RG do Permissionário';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.ORGAO_EMISSOR  IS 'Órgão que emitiu o RG/Identidade do Permissionário';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.SEXO  IS 'Sexo do Permissionário';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.ESTADO_CIVIL  IS 'Estado Civil do Permissionário';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.DATA_NASCIMENTO  IS 'Data de Nascimento do Permissionário';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.UF_PERMISSIONARIO  IS 'UF do Permissionário';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.BAIRRO_PERMISSIONARIO IS 'Bairro do Permissionário';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.ENDERECO_PERMISSIONARIO  IS 'Endereço do Permissionário';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.CELULAR_PERMISSIONARIO  IS 'Celular do Permissionário';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.EMAIL_PERMISSIONARIO  IS 'Email do Permissionário';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.CNH_PERMISSIONARIO  IS 'Habilitação do Permissionário';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.CATEGORIA_CNH_PERMISSIONARIO  IS 'Categoria da CNH do Permissionário';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.NUMERO_QUITACAO_MILITAR  IS 'Número da quitação militar';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.NUMERO_QUITACAO_ELEITORAL  IS 'Número da quitação eleitoral';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.NUMERO_INSCRICAO_INSS  IS 'Número da inscrição no INSS';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.NUMERO_CERTIFICADO_CONDUTOR  IS 'Número do certificado de condutor';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.CERTIDAO_NEGATIVA_CRIMINAL  IS 'Certidão Negativa Criminal do Permissionário';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.CERTIDAO_NEGATIVA_MUNICIPAL  IS 'Certidão Negativa Municipal do Permissionário';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.FOTO  IS 'Foto 3x4 do Permissionário';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.DATA_CRIACAO  IS 'Data de criação do Permissionário';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.STATUS  IS 'Indica O Status: 1-Ativo; 2-Inativo do Permissionário';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.APLICATIVO_ALTERNATIVO  IS 'Indica se o Permissionário é de Aplicativo Alternativo';
COMMENT ON COLUMN PROJ.PERMISSIONARIO.OBSERVACAO  IS 'Observação com dados gerais do Permissionário';
