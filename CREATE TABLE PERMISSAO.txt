CREATE TABLE PROJ.PERMISSAO
( ID_PERMISSAO NUMBER DEFAULT PROJ.SEQ_PERMISSAO.NEXTVAL,
  NUMERO_PERMISSAO VARCHAR2(50) NOT NULL,
  NUMERO_ALVARA VARCHAR2(20),
  ANO_ALVARA VARCHAR2(20),
  CATEGORIA_PERMISSAO VARCHAR2(2) NOT NULL,
  STATUS_PERMISSAO VARCHAR2(2) NOT NULL,
  PERIODO_INICIAL_STATUS DATE NOT NULL,
  PERIODO_FINAL_STATUS DATE,
  DATA_VALIDADE_PERMISSAO DATE NOT NULL,
  PENALIDADE VARCHAR2(2),
  DATA_VALIDADE_PENALIDADE DATE,
  DATA_CRIACAO DATE NOT NULL,
  CONSTRAINT PERMISSAO_PK PRIMARY KEY (ID_PERMISSAO)
);
-- Add comments to the columns 
COMMENT ON COLUMN PROJ.PERMISSAO.ID_PERMISSAO IS 'Identificador único da Permissão';
COMMENT ON COLUMN PROJ.PERMISSAO.NUMERO_PERMISSAO  IS 'Número da Permissão';
COMMENT ON COLUMN PROJ.PERMISSAO.NUMERO_ALVARA  IS 'Número do Alvará';
COMMENT ON COLUMN PROJ.PERMISSAO.ANO_ALVARA  IS 'Ano do Alvará';
COMMENT ON COLUMN PROJ.PERMISSAO.CATEGORIA_PERMISSAO  IS 'Categoria da Permissão';
COMMENT ON COLUMN PROJ.PERMISSAO.STATUS_PERMISSAO IS 'Status da Permissão';
COMMENT ON COLUMN PROJ.PERMISSAO.PERIODO_INICIAL_STATUS  IS 'Período Inicial da Permissão';
COMMENT ON COLUMN PROJ.PERMISSAO.PERIODO_FINAL_STATUS  IS 'Período Final da Permissão';
COMMENT ON COLUMN PROJ.PERMISSAO.DATA_VALIDADE_PERMISSAO  IS 'Data Validade Permissão';
COMMENT ON COLUMN PROJ.PERMISSAO.PENALIDADE  IS 'Pernalidade aplicada a permissão';
COMMENT ON COLUMN PROJ.PERMISSAO.DATA_VALIDADE_PENALIDADE IS 'Data de validade da permissão';
COMMENT ON COLUMN PROJ.PERMISSAO.DATA_CRIACAO  IS 'Data de criação da Permissão';
