CREATE TABLE PROJ.AUDITORIA
( ID_AUDITORIA NUMBER DEFAULT PROJ.SEQ_AUDITORIA.NEXTVAL,
  NOME_MODULO VARCHAR2(200) NOT NULL,
  USUARIO_OPERACAO VARCHAR2(50) NOT NULL,
  OPERACAO VARCHAR2(50) NOT NULL,
  DATA_OPERACAO DATE,
  CONSTRAINT AUDITORIA_PK PRIMARY KEY (ID_AUDITORIA)
);
-- Add comments to the columns 
COMMENT ON COLUMN PROJ.AUDITORIA.ID_AUDITORIA IS 'Identificador único da auditoria';
COMMENT ON COLUMN PROJ.AUDITORIA.NOME_MODULO  IS 'Nome do módulo que foi acessado';
COMMENT ON COLUMN PROJ.AUDITORIA.USUARIO_OPERACAO  IS 'Usuário que realizou a Operação';
COMMENT ON COLUMN PROJ.AUDITORIA.OPERACAO  IS 'Operação realizada: INCLUSÃO, ALTERAÇÃO e EXCLUSÃO';
COMMENT ON COLUMN PROJ.AUDITORIA.DATA_OPERACAO  IS 'Data da Operação';
