CREATE TABLE PROJ.FISCALIZACAO
( ID_FISCALIZACAO NUMBER DEFAULT PROJ.SEQ_FISCALIZACAO.NEXTVAL,	
  ID_VEICULO NUMBER NOT NULL,
  DATA_FISCALIZACAO DATE NOT NULL,
  MOTIVO_INFRACAO VARCHAR2(200) NOT NULL, 
  TIPO_INFRACAO VARCHAR2(2) NOT NULL,  
  GRUPO_MULTAS VARCHAR2(2) NOT NULL,
  PRAZO_REGULARIZACAO DATE NOT NULL,
  NATUREZA_INFRACAO VARCHAR2(2) NOT NULL,
  MODALIDADE VARCHAR2(2) NOT NULL,
  PENALIDADE VARCHAR2(2) NOT NULL,
  OBSERVACAO VARCHAR2(250) NOT NULL,
  DATA_CRIACAO DATE,
  STATUS VARCHAR2(10),
  CONSTRAINT FISCALIZACAO_PK PRIMARY KEY (ID_FISCALIZACAO)
);
-- Add comments to the columns 
COMMENT ON COLUMN PROJ.FISCALIZACAO.ID_FISCALIZACAO IS 'Identificador único da fiscalização';
COMMENT ON COLUMN PROJ.FISCALIZACAO.ID_VEICULO  IS 'Identificador do veículo';
COMMENT ON COLUMN PROJ.FISCALIZACAO.DATA_FISCALIZACAO  IS 'Data da fiscalização';
COMMENT ON COLUMN PROJ.FISCALIZACAO.MOTIVO_INFRACAO  IS 'Motivo da infração';
COMMENT ON COLUMN PROJ.FISCALIZACAO.TIPO_INFRACAO  IS 'Tipo de infração';
COMMENT ON COLUMN PROJ.FISCALIZACAO.GRUPO_MULTAS  IS 'Grupo de multas';
COMMENT ON COLUMN PROJ.FISCALIZACAO.PRAZO_REGULARIZACAO IS 'Prazo de regularização da infração';
COMMENT ON COLUMN PROJ.FISCALIZACAO.NATUREZA_INFRACAO  IS 'Natureza da infração';
COMMENT ON COLUMN PROJ.FISCALIZACAO.MODALIDADE  IS 'Modalidade da infração';
COMMENT ON COLUMN PROJ.FISCALIZACAO.PENALIDADE  IS 'Penalidade da infração';
COMMENT ON COLUMN PROJ.FISCALIZACAO.OBSERVACAO  IS 'Observação para informações adicionais';
COMMENT ON COLUMN PROJ.FISCALIZACAO.DATA_CRIACAO  IS 'Data de criação da fiscalização';
